# siamese L1 Distance layer
class L1Dist(Layer):
    # SỬA Ở ĐÂY: Dùng 2 dấu sao (**)
    def __init__(self, **kwargs):
        # SỬA Ở ĐÂY: Truyền kwargs vào super()
        super().__init__(**kwargs)

    # similarity calculation
    def call(self, input_embedding, validation_embedding):
        return tf.math.abs(input_embedding - validation_embedding)

class AuthScreen(BoxLayout):
    def __init__(self, app, **kwargs):
        super().__init__(**kwargs)
        self.app = app
        self.orientation = 'vertical'
        self.padding = 20
        self.spacing = 20
        
        # Title
        title = Label(
            text='Face Authentication System',
            size_hint=(1, 0.2),
            font_size='24sp',
            bold=True
        )
        
        # Camera widget
        self.camera = Camera(
            resolution=(640, 480),
            play=True,
            size_hint=(1, 0.6)
        )
        
        # Capture button
        self.capture_btn = Button(
            text='Capture & Authenticate',
            size_hint=(1, 0.1),
            background_color=(0.2, 0.6, 0.8, 1),
            font_size='18sp'
        )
        self.capture_btn.bind(on_press=self.capture_and_authenticate)
        
        # Status label
        self.status_label = Label(
            text='Please position your face and click Capture',
            size_hint=(1, 0.1),
            font_size='16sp'
        )
        
        # Add widgets to layout
        self.add_widget(title)
        self.add_widget(self.camera)
        self.add_widget(self.capture_btn)
        self.add_widget(self.status_label)
    
    def capture_and_authenticate(self, instance):
        """Capture image from camera and authenticate"""
        self.status_label.text = "Processing..."
        self.capture_btn.disabled = True
        
        # Schedule the capture to occur after the current frame
        Clock.schedule_once(self.process_capture, 0.1)
    
    def process_capture(self, dt):
        """Process the captured image"""
        try:
            # Get camera texture and convert to OpenCV format
            texture = self.camera.texture
            if texture is None:
                self.status_label.text = "Camera not ready. Please try again."
                self.capture_btn.disabled = False
                return
            
            # Convert texture to numpy array
            buffer = texture.pixels
            image = np.frombuffer(buffer, dtype=np.uint8)
            image = image.reshape(texture.height, texture.width, 4)  # RGBA
            image = cv2.cvtColor(image, cv2.COLOR_RGBA2BGR)
            
            # Preprocess the image for your model
            processed_image = self.app.preprocess_image(image)
            
            # Make prediction
            prediction = self.app.model.predict(
                [self.app.anchor_image, processed_image], 
                verbose=0
            )
            result = prediction[0][0]  # Assuming binary classification
            
            # Check authentication result
            if result >= 0.5:  # Adjust threshold as needed
                self.app.show_success_popup()
            else:
                self.app.show_failure_popup()
                
        except Exception as e:
            Logger.error(f"FaceAuth: Error during authentication: {e}")
            self.status_label.text = "Authentication failed. Please try again."
        
        finally:
            self.capture_btn.disabled = False
    
    

class DashboardScreen(BoxLayout):
    def __init__(self, app, **kwargs):
        super().__init__(**kwargs)
        self.app = app
        self.orientation = 'vertical'
        self.padding = 20
        self.spacing = 20
        
        # Create upload directories if they don't exist
        self.create_upload_directories()
        
        # Title
        title = Label(
            text='Dashboard - File Manager',
            size_hint=(1, 0.1),
            font_size='24sp',
            bold=True
        )
        self.add_widget(title)
        
        # Button section
        self.create_button_section()
        
        # Display area
        self.create_display_section()
        
        # Load initial files
        self.load_files()
    
    def create_upload_directories(self):
        """Create directories for photos and documents"""
        self.photos_dir = Path("uploaded_photos")
        self.documents_dir = Path("uploaded_documents")
        
        self.photos_dir.mkdir(exist_ok=True)
        self.documents_dir.mkdir(exist_ok=True)
    
    def create_button_section(self):
        """Create upload buttons section"""
        button_layout = BoxLayout(orientation='horizontal', size_hint=(1, 0.1), spacing=20)
        
        # Upload Photo button
        upload_photo_btn = Button(
            text='Upload Photo',
            background_color=(0.2, 0.8, 0.2, 1),
            font_size='18sp'
        )
        upload_photo_btn.bind(on_press=self.show_photo_upload)
        
        # Upload Document button
        upload_doc_btn = Button(
            text='Upload Document',
            background_color=(0.2, 0.6, 0.8, 1),
            font_size='18sp'
        )
        upload_doc_btn.bind(on_press=self.show_document_upload)
        
        # Refresh button
        refresh_btn = Button(
            text='Refresh Files',
            background_color=(0.8, 0.8, 0.2, 1),
            font_size='18sp'
        )
        refresh_btn.bind(on_press=self.refresh_files)
        
        button_layout.add_widget(upload_photo_btn)
        button_layout.add_widget(upload_doc_btn)
        button_layout.add_widget(refresh_btn)
        
        self.add_widget(button_layout)
    
    def create_display_section(self):
        """Create file display section"""
        # Main display area
        display_layout = BoxLayout(orientation='vertical', size_hint=(1, 0.8), spacing=10)
        
        # Photos section
        photos_label = Label(
            text='Uploaded Photos:',
            size_hint=(1, 0.05),
            font_size='18sp',
            bold=True
        )
        display_layout.add_widget(photos_label)
        
        self.photos_scroll = ScrollView(size_hint=(1, 0.4))
        self.photos_grid = GridLayout(cols=3, spacing=10, size_hint_y=None)
        self.photos_grid.bind(minimum_height=self.photos_grid.setter('height'))
        self.photos_scroll.add_widget(self.photos_grid)
        display_layout.add_widget(self.photos_scroll)
        
        # Documents section
        documents_label = Label(
            text='Uploaded Documents:',
            size_hint=(1, 0.05),
            font_size='18sp',
            bold=True
        )
        display_layout.add_widget(documents_label)
        
        self.documents_scroll = ScrollView(size_hint=(1, 0.4))
        self.documents_layout = BoxLayout(orientation='vertical', spacing=5, size_hint_y=None)
        self.documents_layout.bind(minimum_height=self.documents_layout.setter('height'))
        self.documents_scroll.add_widget(self.documents_layout)
        display_layout.add_widget(self.documents_scroll)
        
        self.add_widget(display_layout)
    
    def show_photo_upload(self, instance):
        """Show file chooser for photo upload"""
        content = BoxLayout(orientation='vertical', spacing=10)
        
        filechooser = FileChooserListView(
            filters=['*.png', '*.jpg', '*.jpeg', '*.bmp', '*.gif']
        )
        content.add_widget(filechooser)
        
        button_layout = BoxLayout(orientation='horizontal', size_hint=(1, 0.2), spacing=10)
        
        btn_upload = Button(text='Upload Photo', background_color=(0.2, 0.8, 0.2, 1))
        btn_cancel = Button(text='Cancel', background_color=(0.8, 0.2, 0.2, 1))
        
        button_layout.add_widget(btn_upload)
        button_layout.add_widget(btn_cancel)
        content.add_widget(button_layout)
        
        popup = Popup(
            title='Select Photo to Upload',
            content=content,
            size_hint=(0.9, 0.9)
        )
        
        def upload_photo(btn):
            if filechooser.selection:
                source_path = filechooser.selection[0]
                filename = os.path.basename(source_path)
                dest_path = self.photos_dir / filename
                
                try:
                    shutil.copy2(source_path, dest_path)
                    self.load_files()
                    popup.dismiss()
                    self.app.show_message("Photo uploaded successfully!")
                except Exception as e:
                    self.app.show_message(f"Error uploading photo: {e}")
        
        btn_upload.bind(on_press=upload_photo)
        btn_cancel.bind(on_press=popup.dismiss)
        
        popup.open()
    
    def show_document_upload(self, instance):
        """Show file chooser for document upload"""
        content = BoxLayout(orientation='vertical', spacing=10)
        
        filechooser = FileChooserListView(
            filters=['*.pdf', '*.doc', '*.docx', '*.txt', '*.rtf']
        )
        content.add_widget(filechooser)
        
        button_layout = BoxLayout(orientation='horizontal', size_hint=(1, 0.2), spacing=10)
        
        btn_upload = Button(text='Upload Document', background_color=(0.2, 0.6, 0.8, 1))
        btn_cancel = Button(text='Cancel', background_color=(0.8, 0.2, 0.2, 1))
        
        button_layout.add_widget(btn_upload)
        button_layout.add_widget(btn_cancel)
        content.add_widget(button_layout)
        
        popup = Popup(
            title='Select Document to Upload',
            content=content,
            size_hint=(0.9, 0.9)
        )
        
        def upload_document(btn):
            if filechooser.selection:
                source_path = filechooser.selection[0]
                filename = os.path.basename(source_path)
                dest_path = self.documents_dir / filename
                
                try:
                    shutil.copy2(source_path, dest_path)
                    self.load_files()
                    popup.dismiss()
                    self.app.show_message("Document uploaded successfully!")
                except Exception as e:
                    self.app.show_message(f"Error uploading document: {e}")
        
        btn_upload.bind(on_press=upload_document)
        btn_cancel.bind(on_press=popup.dismiss)
        
        popup.open()
    
    def load_files(self):
        """Load and display uploaded files"""
        # Clear existing widgets
        self.photos_grid.clear_widgets()
        self.documents_layout.clear_widgets()
        
        # Load photos
        photo_files = list(self.photos_dir.glob('*'))
        for photo_path in photo_files:
            if photo_path.is_file():
                try:
                    # Create photo thumbnail
                    photo_box = BoxLayout(orientation='vertical', size_hint_y=None, height=150, spacing=5)
                    
                    photo_img = Image(
                        source=str(photo_path),
                        size_hint=(1, 0.8),
                        allow_stretch=True
                    )
                    
                    photo_label = Label(
                        text=photo_path.name,
                        size_hint=(1, 0.2),
                        text_size=(120, None),
                        halign='center'
                    )
                    
                    photo_box.add_widget(photo_img)
                    photo_box.add_widget(photo_label)
                    self.photos_grid.add_widget(photo_box)
                    
                except Exception as e:
                    Logger.error(f"Error loading photo {photo_path}: {e}")
        
        # Load documents
        doc_files = list(self.documents_dir.glob('*'))
        for doc_path in doc_files:
            if doc_path.is_file():
                doc_box = BoxLayout(orientation='horizontal', size_hint_y=None, height=40, spacing=10)
                
                doc_label = Label(
                    text=doc_path.name,
                    size_hint=(0.8, 1),
                    halign='left',
                    text_size=(400, None)
                )
                
                # For documents, we'll just show the filename since we can't display content easily
                doc_box.add_widget(doc_label)
                self.documents_layout.add_widget(doc_box)
        
        # Update grid heights
        self.photos_grid.height = max(150, len(photo_files) * 160)
        self.documents_layout.height = max(40, len(doc_files) * 45)
    
    def refresh_files(self, instance):
        """Refresh the file display"""
        self.load_files()
        self.app.show_message("Files refreshed!")

class FaceAuthApp(App):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.model = None
        self.auth_screen = None
        self.dashboard_screen = None
        self.anchor_image = None
        
    def build(self):
        ANCHOR_PATH = 'data/anchor/anchor.jpg'
        # Load your pre-trained model
        try:
            self.model = load_model(
                'siamesemodel.h5',
                custom_objects={'L1Dist': L1Dist}  # Thêm tham số này
            )
            Logger.info("FaceAuth: Model loaded successfully")

            # 2. Tải và tiền xử lý ảnh gốc (anchor image)
            if not os.path.exists(ANCHOR_PATH):
                msg = f"Anchor image not found at: {ANCHOR_PATH}"
                Logger.error(f"FaceAuth: {msg}")
                self.show_error_popup(msg)
                return Label(text=msg)
            
            anchor_img_raw = cv2.imread(ANCHOR_PATH)
            if anchor_img_raw is None:
                msg = f"Failed to read anchor image from: {ANCHOR_PATH}"
                Logger.error(f"FaceAuth: {msg}")
                self.show_error_popup(msg)
                return Label(text=msg)

            self.anchor_image = self.preprocess_image(anchor_img_raw)
            Logger.info("FaceAuth: Anchor image loaded and preprocessed.")
        except Exception as e:
            Logger.error(f"FaceAuth: Error loading model: {e}")
            self.show_error_popup("Model loading failed")
            return Label(text="Model loading failed")
        
        # Start with authentication screen
        self.show_auth_screen()
        return self.auth_screen
    
    def show_auth_screen(self):
        """Show authentication screen"""
        if self.auth_screen is None:
            self.auth_screen = AuthScreen(self)
        self.root = self.auth_screen
    
    def show_dashboard(self):
        """Show dashboard screen"""
        if self.dashboard_screen is None:
            self.dashboard_screen = DashboardScreen(self)
        self.root = self.dashboard_screen
    
    def show_success_popup(self):
        """Show success popup and transition to dashboard"""
        content = BoxLayout(orientation='vertical', padding=20, spacing=20)
        content.add_widget(Label(
            text='Authentication Successful!',
            font_size='20sp',
            color=(0, 0.7, 0, 1)
        ))
        
        btn = Button(
            text='Continue to Dashboard',
            size_hint=(1, 0.4),
            background_color=(0, 0.7, 0, 1)
        )
        content.add_widget(btn)
        
        popup = Popup(
            title='Success',
            content=content,
            size_hint=(0.7, 0.4),
            auto_dismiss=False
        )
        
        btn.bind(on_press=popup.dismiss)
        btn.bind(on_press=lambda x: self.show_dashboard())
        
        popup.open()
        self.auth_screen.status_label.text = "Authentication successful!"
    
    def show_failure_popup(self):
        """Show failure popup"""
        content = BoxLayout(orientation='vertical', padding=20, spacing=20)
        content.add_widget(Label(
            text='Authentication Failed!',
            font_size='20sp',
            color=(0.9, 0, 0, 1)
        ))
        content.add_widget(Label(
            text='Please try again or contact administrator.',
            font_size='16sp'
        ))
        
        btn = Button(
            text='Try Again',
            size_hint=(1, 0.4),
            background_color=(0.9, 0, 0, 1)
        )
        content.add_widget(btn)
        
        popup = Popup(
            title='Access Denied',
            content=content,
            size_hint=(0.7, 0.5),
            auto_dismiss=False
        )
        
        btn.bind(on_press=popup.dismiss)
        popup.open()
        self.auth_screen.status_label.text = "Authentication failed. Please try again."
        
    def preprocess_image(self, image):
        """
        Preprocess the image for your SNN model
        Adjust this function based on your model's requirements
        """
        try:
            # Resize to match your model's expected input
            resized = cv2.resize(image, (100, 100))  # Adjust size as needed
            
            # Convert to RGB if needed
            rgb_image = cv2.cvtColor(resized, cv2.COLOR_BGR2RGB)
            
            # Normalize pixel values
            normalized = rgb_image.astype('float32') / 255.0
            
            # Add batch dimension
            batched = np.expand_dims(normalized, axis=0)
            
            return batched
            
        except Exception as e:
            Logger.error(f"FaceAuth: Error preprocessing image: {e}")
            raise
    def show_error_popup(self, message):
        """Show error popup"""
        content = BoxLayout(orientation='vertical', padding=20, spacing=20)
        content.add_widget(Label(text=message, font_size='16sp'))
        
        btn = Button(text='OK', size_hint=(1, 0.4))
        content.add_widget(btn)
        
        popup = Popup(
            title='Error',
            content=content,
            size_hint=(0.6, 0.4)
        )
        
        btn.bind(on_press=popup.dismiss)
        popup.open()
    
    def show_message(self, message):
        """Show a simple message popup"""
        content = BoxLayout(orientation='vertical', padding=20, spacing=20)
        content.add_widget(Label(text=message, font_size='16sp'))
        
        btn = Button(text='OK', size_hint=(1, 0.4))
        content.add_widget(btn)
        
        popup = Popup(
            title='Info',
            content=content,
            size_hint=(0.6, 0.3)
        )
        
        btn.bind(on_press=popup.dismiss)
        popup.open()
