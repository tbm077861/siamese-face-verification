import os
from kivy.app import App
from kivy.uix.screenmanager import ScreenManager, Screen
from kivy.uix.boxlayout import BoxLayout
from kivy.uix.gridlayout import GridLayout
from kivy.uix.button import Button
from kivy.uix.label import Label
from kivy.uix.scrollview import ScrollView
from kivy.uix.image import Image
from kivy.core.window import Window
from kivy.uix.filechooser import FileChooserIconView
from kivy.uix.popup import Popup
from kivy.graphics import Color, Rectangle
from pathlib import Path

# Set window size
Window.size = (800, 600)
Window.clearcolor = (0.95, 0.95, 0.97, 1)  # Light gray-blue background

# Define folder paths
PHOTOS_FOLDER = "photos"
DOCUMENTS_FOLDER = "documents"


class LoginScreen(Screen):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        
        layout = BoxLayout(orientation='vertical', padding=50, spacing=20)
        
        # Title
        title = Label(
            text='File Management System',
            font_size='32sp',
            size_hint=(1, 0.3),
            bold=True,
            color=(0.2, 0.3, 0.5, 1)  # Dark blue text
        )
        
        # Login button
        login_btn = Button(
            text='Login',
            size_hint=(0.5, 0.2),
            pos_hint={'center_x': 0.5},
            font_size='20sp',
            background_color=(0.2, 0.6, 0.86, 1),  # Blue button
            color=(1, 1, 1, 1),  # White text
            background_normal=''
        )
        login_btn.bind(on_press=self.go_to_dashboard)
        
        layout.add_widget(title)
        layout.add_widget(login_btn)
        
        self.add_widget(layout)
    
    def go_to_dashboard(self, instance):
        self.manager.current = 'dashboard'


class DashboardScreen(Screen):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.setup_folders()
        self.build_ui()
    
    def setup_folders(self):
        """Create folders if they don't exist"""
        Path(PHOTOS_FOLDER).mkdir(exist_ok=True)
        Path(DOCUMENTS_FOLDER).mkdir(exist_ok=True)
    
    def build_ui(self):
        main_layout = BoxLayout(orientation='vertical', padding=10, spacing=10)
        
        # Header with background
        header = BoxLayout(size_hint=(1, 0.1), spacing=10)
        with header.canvas.before:
            Color(0.2, 0.3, 0.5, 1)  # Dark blue
            header.rect = Rectangle(pos=header.pos, size=header.size)
        header.bind(pos=self.update_rect, size=self.update_rect)
        
        title = Label(
            text='Dashboard',
            font_size='24sp',
            bold=True,
            color=(1, 1, 1, 1)  # White text
        )
        logout_btn = Button(
            text='Logout',
            size_hint=(0.2, 1),
            on_press=self.logout,
            background_color=(0.9, 0.3, 0.3, 1),  # Red button
            color=(1, 1, 1, 1),  # White text
            background_normal=''
        )
        header.add_widget(title)
        header.add_widget(logout_btn)
        
        # Content area with two sections
        content = BoxLayout(orientation='horizontal', spacing=10, size_hint=(1, 0.9))
        
        # Photos section
        photos_section = self.create_section('Photos', PHOTOS_FOLDER, is_image=True)
        
        # Documents section
        docs_section = self.create_section('Documents', DOCUMENTS_FOLDER, is_image=False)
        
        content.add_widget(photos_section)
        content.add_widget(docs_section)
        
        main_layout.add_widget(header)
        main_layout.add_widget(content)
        
        self.add_widget(main_layout)
    
    def update_rect(self, instance, value):
        """Update rectangle size/position when widget changes"""
        instance.rect.pos = instance.pos
        instance.rect.size = instance.size
    
    def create_section(self, title, folder_path, is_image):
        """Create a section for photos or documents"""
        section = BoxLayout(orientation='vertical', spacing=5)
        
        # Add white background to section
        with section.canvas.before:
            Color(1, 1, 1, 1)  # White background
            section.rect = Rectangle(pos=section.pos, size=section.size)
        section.bind(pos=self.update_rect, size=self.update_rect)
        
        # Section header
        header = BoxLayout(size_hint=(1, 0.1), spacing=5, padding=10)
        title_label = Label(
            text=title,
            font_size='18sp',
            bold=True,
            color=(0.2, 0.3, 0.5, 1)  # Dark blue text
        )
        upload_btn = Button(
            text=f'Upload {title}',
            size_hint=(0.5, 1),
            on_press=lambda x: self.open_file_chooser(folder_path, is_image),
            background_color=(0.3, 0.7, 0.3, 1),  # Green button
            color=(1, 1, 1, 1),  # White text
            background_normal=''
        )
        header.add_widget(title_label)
        header.add_widget(upload_btn)
        
        # File display area
        scroll = ScrollView(size_hint=(1, 0.9))
        file_grid = GridLayout(cols=2, spacing=10, size_hint_y=None, padding=10)
        file_grid.bind(minimum_height=file_grid.setter('height'))
        
        # Load existing files
        self.load_files(folder_path, file_grid, is_image)
        
        scroll.add_widget(file_grid)
        
        section.add_widget(header)
        section.add_widget(scroll)
        
        # Store references for updating
        if is_image:
            self.photos_grid = file_grid
        else:
            self.docs_grid = file_grid
        
        return section
    
    def load_files(self, folder_path, grid, is_image):
        """Load files from folder and display them"""
        grid.clear_widgets()
        
        folder = Path(folder_path)
        files = list(folder.iterdir())
        
        if not files:
            no_files = Label(
                text=f'No files yet\nClick "Upload" to add files',
                size_hint_y=None,
                height=100,
                color=(0.5, 0.5, 0.5, 1)  # Gray text
            )
            grid.add_widget(no_files)
            return
        
        for file_path in files:
            if file_path.is_file():
                file_widget = BoxLayout(
                    orientation='vertical',
                    size_hint_y=None,
                    height=150,
                    spacing=5
                )
                
                if is_image and file_path.suffix.lower() in ['.png', '.jpg', '.jpeg', '.gif', '.bmp']:
                    img = Image(
                        source=str(file_path),
                        size_hint=(1, 0.8),
                        allow_stretch=True,
                        keep_ratio=True
                    )
                    file_widget.add_widget(img)
                else:
                    file_icon = Label(
                        text=f'📄\n{file_path.suffix}',
                        size_hint=(1, 0.8),
                        font_size='24sp',
                        color=(0.3, 0.3, 0.3, 1)  # Dark gray
                    )
                    file_widget.add_widget(file_icon)
                
                file_name = Label(
                    text=file_path.name,
                    size_hint=(1, 0.2),
                    font_size='10sp',
                    text_size=(None, None),
                    shorten=True,
                    shorten_from='right',
                    color=(0.2, 0.2, 0.2, 1)  # Almost black
                )
                file_widget.add_widget(file_name)
                
                grid.add_widget(file_widget)
    
    def open_file_chooser(self, folder_path, is_image):
        """Open file chooser dialog"""
        content = BoxLayout(orientation='vertical', spacing=10)
        
        file_chooser = FileChooserIconView(
            size_hint=(1, 0.9),
            filters=['*.*']
        )
        
        buttons = BoxLayout(size_hint=(1, 0.1), spacing=10)
        select_btn = Button(
            text='Select',
            background_color=(0.2, 0.6, 0.86, 1),  # Blue
            color=(1, 1, 1, 1),
            background_normal=''
        )
        cancel_btn = Button(
            text='Cancel',
            background_color=(0.7, 0.7, 0.7, 1),  # Gray
            color=(1, 1, 1, 1),
            background_normal=''
        )
        
        buttons.add_widget(select_btn)
        buttons.add_widget(cancel_btn)
        
        content.add_widget(file_chooser)
        content.add_widget(buttons)
        
        popup = Popup(
            title=f'Choose file to upload',
            content=content,
            size_hint=(0.9, 0.9)
        )
        
        def select_file(instance):
            if file_chooser.selection:
                self.upload_file(file_chooser.selection[0], folder_path, is_image)
                popup.dismiss()
        
        select_btn.bind(on_press=select_file)
        cancel_btn.bind(on_press=popup.dismiss)
        
        popup.open()
    
    def upload_file(self, source_path, dest_folder, is_image):
        """Copy file to destination folder"""
        import shutil
        
        source = Path(source_path)
        destination = Path(dest_folder) / source.name
        
        try:
            shutil.copy2(source, destination)
            
            # Refresh the appropriate grid
            if is_image:
                self.load_files(dest_folder, self.photos_grid, True)
            else:
                self.load_files(dest_folder, self.docs_grid, False)
            
            # Show success message
            popup = Popup(
                title='Success',
                content=Label(text=f'File uploaded successfully!'),
                size_hint=(0.5, 0.3)
            )
            popup.open()
        except Exception as e:
            popup = Popup(
                title='Error',
                content=Label(text=f'Failed to upload file:\n{str(e)}'),
                size_hint=(0.5, 0.3)
            )
            popup.open()
    
    def logout(self, instance):
        self.manager.current = 'login'


class FileManagementApp(App):
    def build(self):
        sm = ScreenManager()
        sm.add_widget(LoginScreen(name='login'))
        sm.add_widget(DashboardScreen(name='dashboard'))
        return sm


if __name__ == '__main__':
    FileManagementApp().run()
